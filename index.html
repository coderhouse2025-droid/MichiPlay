// 🧶 Posición dinámica del ovillo (respawn)
function respawnYarn() {
    if(gameOver) return;
    yarnBall.collected = false;
    // Aparece en alguna plataforma aleatoria
    let safePlats = platforms.filter(p => p.y < 380);
    let rand = safePlats[Math.floor(Math.random() * safePlats.length)];
    yarnBall.x = rand.x + rand.w/2;
    yarnBall.y = rand.y - 8;
    yarnBall.radius = 9;
}
